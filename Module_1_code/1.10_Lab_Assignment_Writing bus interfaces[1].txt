RTL Design Code:

`timescale 1ns/1ps
module apb_timer #(
    parameter WIDTH = 8 // Width of counter register
)( 
//Standard APB interface signals
    input  wire        PCLK, //APB clock
    input  wire        PRESETn, // Active-low reset
    input  wire        PSEL, //Peripheral select
    input  wire        PENABLE, // Transfer enable
    input  wire        PWRITE, //1 = write, 0 = read
    input  wire [7:0]  PADDR, // Register address
    input  wire [31:0] PWDATA, //Write data from bus
    output reg  [31:0] PRDATA, // Read data to bus


// Peripheral output
    output reg         timer_done // signals when timer finishes
);


//Internal registers for timer logic
    reg [WIDTH-1:0] load_val; //Preset countdown value
    reg [WIDTH-1:0] count; //Current counter value
    reg             running; //Indicates if timer is active

    // Mark unused upper bits of PWDATA as used
    wire unused_pwd = |PWDATA[31:WIDTH];
//-------------------
	// APB Write logic
	//-------------------
	// Awrite occurs during the Access phase: PSEL=1, PENABLE=1,
	// PWRITE= 1

    always @(posedge PCLK or negedge PRESETn) begin
        if (!PRESETn) begin
            load_val <= 0;
            running  <= 0;
        end
        else if (PSEL && PENABLE && PWRITE) begin
            case (PADDR)
                8'h00: load_val <= PWDATA[WIDTH-1:0]; //Write load value
                8'h04: running  <= PWDATA[0]; //start/stop control
                
            endcase
        end
    end

//------------------------------------
	//APBRead Logic
	//------------------------------------
	// Start/stop control 
	//Onreads (PWRITE=0), PRDATA outputs register values

    always @(*) begin
        PRDATA = 32'h0; // Default
        if (PSEL && !PWRITE) begin
            case (PADDR)
                8'h00: PRDATA = {{(32-WIDTH){1'b0}}, load_val}; // Read load value
                8'h04: PRDATA = {31'b0, running}; // Read status
                8'h08: PRDATA = {31'b0, timer_done}; // Read status
                default: PRDATA = 32'h0;
            endcase
        end
    end

//------------
	//Timer Countdown logic
	//----------------------
	//Decrements counter until zero, then sets timer_done
    always @(posedge PCLK or negedge PRESETn) begin
        if (!PRESETn) begin
            count      <= 0;
            timer_done <= 0;
        end
        else if (running) begin
            if (count == 0) begin
                count      <= load_val; // Reload for one-shot
                timer_done <= 1; // Signal done
                running    <= 0;
            end
            else begin
                count      <= count - 1; // Decrement counter
                timer_done <= 0; //Clear done until next cycle
            end
        end
        
    end

endmodule
		 


Step 3: Create Testbench File
3.1: Open gvim editor for testbench with command: gvim apb_timer_tb.v

Testbench:
// Testbench for APB Timer peripheral

`timescale 1ns/1ps
module apb_timer_tb;

//APB signals
    reg         PCLK;
    reg         PRESETn = 0;
    reg         PSEL = 0;
    reg         PENABLE = 0;
    reg         PWRITE = 0;
    reg  [7:0]  PADDR = 0;
    reg  [31:0] PWDATA = 0;
    wire [31:0] PRDATA;
    wire        timer_done;

// Instantiate DUT
    apb_timer uut (
        .PCLK(PCLK),
        .PRESETn(PRESETn),
        .PSEL(PSEL),
        .PENABLE(PENABLE),
        .PWRITE(PWRITE),
        .PADDR(PADDR),
        .PWDATA(PWDATA),
        .PRDATA(PRDATA),
        .timer_done(timer_done)
    );

// clock generation (10nsperiod)
    initial PCLK = 0;
    always #5 PCLK = ~PCLK;


    // Use timer_done to avoid unused warning
    always @(posedge timer_done)
        $display("Time=%0t : TIMER DONE", $time);


// ------------------------
// APB Write Task
// -------------------------------
// Simulates a bus write transaction


    task apb_write(input [7:0] addr, input [31:0] data);
    begin
        @(posedge PCLK);
        PSEL=1; PWRITE=1; PENABLE=0; PADDR=addr; PWDATA=data;
        @(posedge PCLK);
        PENABLE=1;
        @(posedge PCLK);
        PSEL=0; PENABLE=0; PWRITE=0;
    end
    endtask



//-----------------------
// APB Read Task
// ----------------
// Simulates a bus read transaction

    task apb_read(input [7:0] addr);
    begin
        @(posedge PCLK);
        PSEL=1; PWRITE=0; PENABLE=0; PADDR=addr;
        @(posedge PCLK);
        PENABLE=1;
        @(posedge PCLK);
        $display("Read [0x%0h] = %0d", addr, PRDATA);
        PSEL=0; PENABLE=0;
    end
    endtask

//---------------
// Test Sequence
// ---------------------

    initial begin
        $dumpfile("apb_timer.vcd");
        $dumpvars(0, apb_timer_tb);

	// step 1: Reset the design
        #10 PRESETn = 1;

	// step 2: Write load value = 5
        apb_write(8'h00, 5);

	// step 3: Start timer
        apb_write(8'h04, 1);

	// step 4: Wait and read status register
        #80 apb_read(8'h08);

	// step 5: Write lad value = 3
        apb_write(8'h00, 3); 

	// step 6: Start timer again
        apb_write(8'h04, 1);

	// step 7: Read status after short delay
        #50 apb_read(8'h08);

        #50 $finish;
    end

endmodule


3.2: Save the testbench file using command: :wq!
3.3: Check the saved file in current directory: use command ls

Step 4: Simulate the Design with Verilator


4.1: Use the following command to simulate:
verilator --binary -j 0 -Wall apb_timer.v apb_timer_tb.v --top apb_timer_tb --timing --CFLAGS "-std=c++20" --trace
4.2: If any error is available, resolve it.
4.3: After simulation, a new directory obj_dir will be created. Check with ls.
4.4: Open obj_dir: cd obj_dir
4.5: Use command make -f Vapb_timer_tb.mk Vapb_timer_tb
4.6: A new executable named Vapb_timer_tb will be created
4.7: Use command ./Vapb_timer_tb to run simulation
4.8: Check output in terminal
