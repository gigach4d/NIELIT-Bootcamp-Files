Inserting Clock and Reset Synchronization Logic

1: 2-Stage Reset Synchronizer 

1.1 ) Design file for reset synchronizer: reset_sync.v
// Converts an asynchronous reset into a safe, glitch-free synchronous reset

module reset_sync (
  input clk,              // Destination clock domain
  input async_reset,     // Asynchronous reset input
  output reg sync_reset   // Synchronized reset output
);
  reg sync_ff1;          // First flip-flop for capturing async_reset

// Use both posedge clk and posedge async_reset for immediate assertion
  always @(posedge clk or posedge async_reset) begin
    if (async_reset) begin
      // Immediate assertion of reset in both flops
      sync_ff1     <= 1'b1;
      sync_reset   <= 1'b1;
    end else begin
      // Deassertion happens synchronously across clock edges
      sync_ff1     <= 1'b0;
      sync_reset   <= sync_ff1;
    end
  end
endmodule

2: 2-Stage Data Synchronizer (CDC Logic) 

2.1 ) Design file for 2-stage Data Synchronizer: cdc_sync.v
// Safely captures an asynchronous data input into the clk domain
module cdc_sync (
  input clk,             // Destination clock
  input async_in,        // Asynchronous input signal
  output reg sync_out    // Synchronized output
);
  reg stage1;            // Intermediate register
 
 // 2-stage flip-flop synchronizer
  always @(posedge clk) begin
    stage1   <= async_in;     // First flop captures async signal
    sync_out <= stage1;       // Second flop settles metastability
  end
endmodule

3: Integration of Reset and Signal Synchronizers 

3.1) Design file name : top_with_sync.v
// Top module that integrates reset and CDC synchronizers
module top_with_sync (
  input clk,              // Main clock input
  input async_reset,      // Global asynchronous reset
  input async_signal,     // CDC input from another domain
  output synced_reset,    // Cleaned reset
  output synced_signal    // Cleaned signal
);

  // Instantiate the reset synchronizer
  reset_sync u_reset_sync (
    .clk(clk),
    .async_reset(async_reset),
    .sync_reset(synced_reset)
  );

  // Instantiate the 2-stage CDC synchronizer
  cdc_sync u_cdc_sync (
    .clk(clk),
    .async_in(async_signal),
    .sync_out(synced_signal)
  );

endmodule

4: Testbench to Verify Synchronization Logic

Testbench file : sync_tb.v
// Testbench to verify reset and CDC synchronization behavior
module sync_tb;

  reg clk = 0;               // System clock
  reg async_reset = 1;       // Start with reset active
  reg async_signal = 0;      // Default async input
  wire synced_reset;
  wire synced_signal;

  // Generate 10ns clock
  always #5 clk = ~clk;

  // Instantiate DUT
  top_with_sync dut (
    .clk(clk),
    .async_reset(async_reset),
    .async_signal(async_signal),
    .synced_reset(synced_reset),
    .synced_signal(synced_signal)
  );

  initial begin
    $dumpfile("dump.vcd");
    $dumpvars(0, sync_tb);

    // Deassert reset after some time
    #12 async_reset = 0;

    // Inject asynchronous signal pulses
    #15 async_signal = 1;
    #3  async_signal = 0;

    #20 async_signal = 1;
    #2  async_signal = 0;

    #40 $finish;
  end

endmodule


Create Simulation script 

a) Create “run.sh”

#!/bin/bash
# Step 1: Run Verilator to compile and generate simulation files
verilator --binary -j 0 -Wall \
  reset_sync.v cdc_sync.v top_with_sync.v sync_tb.v \
  --top sync_tb --timing --CFLAGS "-std=c++20" --trace
# Step 2: Navigate to the object directory
cd obj_dir || { echo "obj_dir not found"; exit 1; }
# Step 3: Compile the simulation binary
make -f Vsync_tb.mk Vsync_tb || { echo "Compilation failed"; exit 1; }
# Step 4: Run the simulation
./Vsync_tb || { echo "Simulation failed"; exit 1; }
# Step 5: Open the waveform in GTKWave
gtkwave dump.vcd


b) Make the script executable:
chmod +x run.sh
c) Run the script (./run.sh) to perform compilation, simulation, and open the waveform.
