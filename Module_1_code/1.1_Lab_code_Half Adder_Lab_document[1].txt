// 1) Half Adder - Dataflow Modelling
// This model uses continuous assignment with assign statements
`timescale 1ns/1ps  // Time unit = 1ns, time precision = 1ps
module half_adder_dataflow(
  input a,               // First input bit
  input b,               // Second input bit
  output sum,            // Sum output
  output carry           // Carry output
);
  assign sum = a ^ b;    // XOR operation for sum
  assign carry = a & b;  // AND operation for carry

endmodule

// 1) Testbench for Half Adder - Dataflow Modeling
`timescale 1ns/1ps  // Time unit = 1ns, time precision = 1ps
module half_adder_dataflow_tb;
  reg a, b;                 // Input test signals
  wire sum, carry;          // Outputs from the half adder
  // Instantiate Unit Under Test (UUT)
  half_adder_dataflow uut (
    .a(a),
    .b(b),
    .sum(sum),
    .carry(carry)
  );

  // Test stimulus block
  initial begin
    $display("Dataflow Half Adder Test");
    $display("A B | Sum Carry");
    $display("---------------------");

    a = 0; b = 0; #10 $display("%b %b |  %b    %b", a, b, sum, carry);  // 0 + 0
    a = 0; b = 1; #10 $display("%b %b |  %b    %b", a, b, sum, carry);  // 0 + 1
    a = 1; b = 0; #10 $display("%b %b |  %b    %b", a, b, sum, carry);  // 1 + 0
    a = 1; b = 1; #10 $display("%b %b |  %b    %b", a, b, sum, carry);  // 1 + 1

    $finish;  // End simulation
  end

  // Dump waveform data for GTKWave
  initial begin
    $dumpfile("half_adder_dataflow.vcd");  // Output VCD file
    $dumpvars(0, half_adder_dataflow_tb);  // (Dump all variables in testbench)
  end
endmodule

// 2) Half Adder - Behavioral Modeling
// This model uses an always block and procedural assignment
`timescale 1ns/1ps  // Time unit = 1ns, time precision = 1ps
module half_adder_behavioral(
  input a,         // First input bit
  input b,         // Second input bit
  output reg sum,  // Sum output (a XOR b)
  output reg carry // Carry output (a AND b)
);
  //Combinational logic block triggered on changes to 'a' or 'b'
  always @(a or b) begin
    sum = a ^ b;      // XOR for sum
    carry = a & b;    // AND for carry
  end
endmodule

// Half Adder - Structural Modeling
// This model instantiates basic gates to build functionality
`timescale 1ns/1ps  // Time unit = 1ns, time precision = 1ps
module half_adder_structural(
  input a,             // First input bit
  input b,             // Second input bit
  output sum,          // Sum output
  output carry         // Carry output
);

  // Instantiate primitive gates with labels
  xor u1(sum, a, b);    // sum = a XOR b
  and u2(carry, a, b);  // carry = a AND b

endmodule
