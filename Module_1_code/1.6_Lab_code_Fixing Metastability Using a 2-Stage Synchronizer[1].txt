Create 2-stage synchronizer design file: cdc_sync.v

// 2-stage synchronizer module to handle asynchronous input
module cdc_synchronizer (
  input clk,              // Clock input
  input async_in,         // Asynchronous input signal
  output reg synced       // Synchronized output signal
);
  reg stage1;             // First-stage flip-flop
  // Two-stage flip-flop chain for synchronization
  always @(posedge clk) begin
    stage1 <= async_in;
    synced <= stage1;
  end
endmodule


Step 3: Create Testbench File

3.1 Create testbench file : gvim cdc_sync_tb.v:

// Testbench for CDC synchronizer
module cdc_sync_tb;
  reg clk = 0;           // Clock signal
  reg async_in = 0;      // Asynchronous input
  wire synced;           // Output from synchronizer
  // Clock generation: 10ns period
  always #5 clk = ~clk;
  // Instantiate the DUT
  cdc_synchronizer uut (
    .clk(clk),
    .async_in(async_in),
    .synced(synced)
  );
  // Apply transitions not aligned to clk
  initial begin
    $dumpfile("dump.vcd");
    $dumpvars(0, cdc_sync_tb);
    #12 async_in = 1;
    #7  async_in = 0;
    #6  async_in = 1;
    #9  async_in = 0;
    #50 $finish;
  end
endmodule

./Vcdc_sync_tb || { echo "Simulation failed"; exit 1; }
