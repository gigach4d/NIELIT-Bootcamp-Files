// Create RTL Design File

//Create design file: gvim metastability.v

// Module to sample async input directly without synchronization
module metastability (
  input clk,         // Clock input
  input async_in,    // Asynchronous input signal
  output reg sampled //Sampled output (may go metastable)
);
  always @(posedge clk) begin
    sampled <= async_in;
  end
endmodule


// Testbench File

//Create testbench file : gvim metastability_tb.v

// Testbench for metastability 
module metastability_tb;
  reg clk = 0;
  reg async_in = 0;
  wire sampled;
  always #5 clk = ~clk;
  // Device Under Test instantiation
  metastability uut (
    .clk(clk),
    .async_in(async_in),
    .sampled(sampled)
  );
  initial begin
    $dumpfile("dump.vcd");
    $dumpvars(0, metastability_tb);
    #12 async_in = 1;   // Change async_in near clock edge
    #7  async_in = 0;
    #6  async_in = 1;
    #9  async_in = 0;
    #50 $finish;
  end
endmodule


// Create Simulation Script

// Create run.sh

#!/bin/bash
# 1: Run Verilator to generate simulation files
verilator --binary -j 0 -Wall metastability.v metastability_tb.v --top metastability_tb --timing --CFLAGS "-std=c++20" --trace
2: Move into the object directory
cd obj_dir || { echo "obj_dir not found"; exit 1; }
 3: Compile the testbench using make
make -f Vmetastability_tb.mk Vmetastability_tb || { echo "Compilation failed"; exit 1; }

 4: Run the testbench simulation
./Vmetastability_tb || { echo "Simulation failed"; exit 1; }
5: View the waveform using GTKWave
gtkwave dump.vcd


// Make the script executable:
chmod +x run.sh

// Run And Observe The Output 
// Execute the simulation script:  ./run.sh
