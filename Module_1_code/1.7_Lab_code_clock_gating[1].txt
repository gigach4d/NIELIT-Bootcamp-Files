Implementation of Clock Gating

2.1: Regular Flip-Flop without Gating (ff_no_gating.v)

module ff_no_gating (
  input clk,
  input rst,
  input d,
  output reg q
);
always @(posedge clk) begin
    if (rst)
      q <= 0;
    else
      q <= d;
  end
endmodule

2.2: Flip-Flop with Clock Gating (ff_with_gating.v)

module ff_with_gating (
  input clk,
  input rst,
  input d,
  input enable,
  output reg q
);
  wire gated_clk;
  assign gated_clk = clk & enable;
  always @(posedge gated_clk) begin
    if (rst)
      q <= 0;
    else
      q <= d;
end
endmodule

Step 3: Create Testbench File

3.1: Testbench (ff_tb.v)
module ff_tb;
  reg clk = 0, rst = 0, d = 0, enable = 0;
  wire q_nogate, q_gate;
  always #5 clk = ~clk;
  ff_no_gating    u1 (.clk(clk),      .rst(rst), .d(d),       .q(q_nogate));
  ff_with_gating  u2 (.clk(clk), .enable(enable), .rst(rst), .d(d), .q(q_gate));
  initial begin
    $dumpfile("dump.vcd");
    $dumpvars(0, ff_tb);
    rst = 1; d = 0; enable = 0; #10 rst = 0;
    d = 1; #20;
    enable = 1; #30;
    enable = 0; d = 0; #30;
    $finish;
  end
endmodule

#!/bin/bash

cd obj_dir || { echo "obj_dir not found"; exit 1; }

./Vff_tb || { echo "Simulation failed"; exit 1; }
gtkwave dump.vcd
Make the script executable:
chmod +x run.sh

run the script : ./run.sh
