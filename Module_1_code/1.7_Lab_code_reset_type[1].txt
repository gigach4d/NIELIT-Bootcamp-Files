1) Asynchronous Reset
//===========================================================================
1.1) Asynchronous reset design 
Create the files for asynchronous_reset : async_reset_ff.v

module async_reset_ff (
  input clk,        // Clock input
  input rst_n,      // Active-low asynchronous reset
  input d,          // Input data
  output reg q      // Output
);
  
 // Reset has priority and is sampled on negative edge
  always @(posedge clk or negedge rst_n) begin
    if (!rst_n)
      q <= 0;       // Reset: output goes to 0 immediately
    else
      q <= d;
  end
endmodule

1.3 Create testbench for asynchronous reset: async_reset_tb.v

module async_reset_tb;
  reg clk = 0, rst_n = 1, d = 0;
  wire q;
  async_reset_ff uut (.clk(clk), .rst_n(rst_n), .d(d), .q(q));
  always #5 clk = ~clk;
  initial begin
    $dumpfile("async_reset.vcd");
    $dumpvars(0, async_reset_tb);
    d = 1; #8;
    rst_n = 0;   // Trigger async reset mid-cycle
    #4;
    rst_n = 1;
    #20;
    $finish;
  end
endmodule

2) Synchronous Reset
Objective of this is to observe how reset only takes effect at clock edges, even if asserted earlier.
2.1) Synchronous reset design verilog code
Create the files for synchronou_reset : sync_reset_ff.v
module sync_reset_ff (
  input clk,
  input rst,        // Active-high reset
  input d,
  output reg q
);
  always @(posedge clk) begin
    if (rst)
      q <= 0;
    else
      q <= d;
  end
endmodule
2.3 Create the file for testbench: sync_reset_tb.v
module sync_reset_tb;
  reg clk = 0, rst = 0, d = 0;
  wire q;
  sync_reset_ff uut (.clk(clk), .rst(rst), .d(d), .q(q));
  always #5 clk = ~clk;
  initial begin
    $dumpfile("sync_reset.vcd");
    $dumpvars(0, sync_reset_tb);
    d = 1; #3;
    rst = 1; #7;
    rst = 0; #20;
    $finish;
  end
endmodule

3: Global Reset
Objective of this is to show how a single reset signal can reset multiple flip-flops or blocks.

Create the files for global_reset : global_reset.v
module global_reset (
  input clk,
  input global_rst,
  input d1, d2,
  output reg q1, q2
);
  always @(posedge clk) begin
    if (global_rst) begin
      q1 <= 0;
      q2 <= 0;
    end else begin
      q1 <= d1;
      q2 <= d2;
    end
  end
endmodule
3.3 :Create a file for global reset testbench : global_reset_tb.v
module global_reset_tb;
  reg clk = 0, global_rst = 0;

  wire q1, q2;

  always #5 clk = ~clk;

    $dumpfile("global_reset.vcd");
    $dumpvars(0, global_reset_tb);

    #10 global_rst = 1;
    #10 global_rst = 0;
    #20 $finish;
  end
endmodule

4) Domain Specific Reset
Objective of this is to understand how subsystems use independent clocks and resets.
4.1) Domain Specific reset design
 Create the file for domain_specific_reset : domain_reset.v
module domain_reset (
  input clk1, clk2,
  input rst1, rst2,
  input d1, d2,
  output reg q1, q2
);
  always @(posedge clk1) begin
    if (rst1)
      q1 <= 0;
    else
      q1 <= d1;
  end
  always @(posedge clk2) begin
    if (rst2)
      q2 <= 0;
    else
      q2 <= d2;
  end
endmodule
4.3 Create the file for domain_testbench : domain_reset_tb.v
module domain_reset_tb;
  reg clk1 = 0, clk2 = 0;
  reg rst1 = 0, rst2 = 0;
  reg d1 = 1, d2 = 1;
  wire q1, q2;

  domain_reset uut (.clk1(clk1), .clk2(clk2), .rst1(rst1), .rst2(rst2), .d1(d1), .d2(d2), .q1(q1), .q2(q2));

  always #4 clk1 = ~clk1; // 8ns clock
  always #6 clk2 = ~clk2; // 12ns clock

  initial begin
    $dumpfile("domain_reset.vcd");
    $dumpvars(0, domain_reset_tb);

    #10 rst1 = 1; #8; rst1 = 0;
    #15 rst2 = 1; #10; rst2 = 0;
    #30 $finish;
  end
endmodule
