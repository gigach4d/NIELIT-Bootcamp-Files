Multi-Stage Synchronizer for Stable CDC
This lab demonstrates how multi-stage flip-flop synchronizers, such as 2-stage and 3-stage designs, effectively mitigate metastability when asynchronous signals cross from one clock domain to another.


Create the design file : sync2_stage.v

// Description: Two flip-flop stages to synchronize async input
module sync2_stage (
  input  wire clk,
  input  wire async_in,
  output reg  sync_out
);
  reg sync_ff1;
  always @(posedge clk) begin
    sync_ff1  <= async_in;   // First stage
    sync_out  <= sync_ff1;   // Second stage
  end
endmodule

// 3-Stage Synchronizer

Create the design file: sync3_stage.v
// Three flip-flop stages for higher robustness
module sync3_stage (
  input  wire clk,
  input  wire async_in,
  output reg  sync_out
);
  reg sync_ff1, sync_ff2;
  always @(posedge clk) begin
    sync_ff1  <= async_in;
    sync_ff2  <= sync_ff1;
    sync_out  <= sync_ff2;
  end
endmodule


2.3: Create Testbench for Both Synchronizers

module sync_tb;
  reg clk = 0;
  reg async_in = 0;
  wire sync_out2, sync_out3;
  always #5 clk = ~clk;
  // Instantiate 2-stage synchronizer
  sync2_stage u_sync2 (
    .clk(clk),
    .async_in(async_in),
    .sync_out(sync_out2)
  );
  // Instantiate 3-stage synchronizer
  sync3_stage u_sync3 (
    .clk(clk),
    .async_in(async_in),
    .sync_out(sync_out3)
  );
  initial begin
    $dumpfile("dump.vcd");
    $dumpvars(0, sync_tb);
    // Toggle async_in at non-clock intervals
    #7  async_in = 1;
    #13 async_in = 0;
    #8  async_in = 1;
    #12 async_in = 0;
    #50 $finish;
  end
endmodule


Step 3: Create the Simulation Script

// Create the simulation script file: run.sh

#!/bin/bash
# 1: Use Verilator to compile and generate simulation files
verilator --binary -j 0 -Wall sync2_stage.v sync3_stage.v sync_tb.v --top sync_tb --timing --CFLAGS "-std=c++20" --trace
# 2: Move to the object directory
cd obj_dir || { echo "obj_dir not found"; exit 1; }
# 3: Compile the simulation binary using make
make -f Vsync_tb.mk Vsync_tb || { echo "Compilation failed"; exit 1; }
# 4: Run the simulation and check for failure
./Vsync_tb || { echo "Simulation failed"; exit 1; }
# 5: Open the waveform in GTKWave
gtkwave dump.vcd
