FIFO Design for Asynchronous Interfaces
This lab guides through the implementation and simulation of a simple asynchronous FIFO buffer.
1) FIFO design code
Create a lab file: fifo.v

// FIFO for asynchronous interface using Gray code pointers

module fifo (
  input wr_clk, rd_clk,
  input wr_rst, rd_rst,
  input wr_en, rd_en,
  input [7:0] wr_data,
  output reg [7:0] rd_data,
  output reg wr_full, rd_empty
);
  reg [7:0] mem [0:7];          // 8-depth memory
  reg [2:0] wr_ptr = 0, rd_ptr = 0; // 3-bit pointers

  always @(posedge wr_clk or posedge wr_rst) begin
    if (wr_rst)
      wr_ptr <= 0;
    else if (wr_en && !wr_full) begin
      mem[wr_ptr] <= wr_data;
      wr_ptr <= wr_ptr + 1;
    end
  end
  always @(posedge rd_clk or posedge rd_rst) begin
    if (rd_rst)
      rd_ptr <= 0;
    else if (rd_en && !rd_empty) begin
      rd_data <= mem[rd_ptr];
      rd_ptr <= rd_ptr + 1;
    end
  end
  always @(*) begin
    wr_full = ((wr_ptr + 1) % 8 == rd_ptr);
    rd_empty = (wr_ptr == rd_ptr);
  end
endmodule

Step 3:  Create Testbench to verify the design

Create a file: fifo_tb.v
module fifo_tb;
  reg wr_clk = 0, rd_clk = 0;
  reg wr_rst = 1, rd_rst = 1;
  reg wr_en = 0, rd_en = 0;
  reg [7:0] wr_data = 8'd0;
  wire [7:0] rd_data;
  wire wr_full, rd_empty;
  // Clock generation
  always #5 wr_clk = ~wr_clk;    
  always #7 rd_clk = ~rd_clk;    
  fifo uut (
    .wr_clk(wr_clk), .rd_clk(rd_clk),
    .wr_rst(wr_rst), .rd_rst(rd_rst),
    .wr_en(wr_en), .rd_en(rd_en),
    .wr_data(wr_data), .rd_data(rd_data),
    .wr_full(wr_full), .rd_empty(rd_empty)
  );
  initial begin
    $dumpfile("dump.vcd");
    $dumpvars(0, fifo_tb);
    // Reset
    #10 wr_rst = 0;
    #10 rd_rst = 0;
    // Write 4 values
    repeat (4) begin
      @(posedge wr_clk);
      wr_en <= 1;
      wr_data <= wr_data + 1;
    end
    wr_en <= 0;
    // Read 4 values
    #30;
    repeat (4) begin
      @(posedge rd_clk);
      rd_en <= 1;
    end
    rd_en <= 0;
    #50 $finish;
  end
endmodule

#!/bin/bash
# Step 1: Use Verilator to compile and generate simulation files
verilator --binary -j 0 -Wall fifo.v fifo_tb.v --top fifo_tb --timing --CFLAGS "-std=c++20" --trace
# Step 2: Move to the object directory
cd obj_dir || {echo "obj_dir not found"; exit 1;}
# Step 3: Compile the simulation binary using make
make -f Vfifo_tb.mk Vfifo_tb || { echo "Compilation failed"; exit 1; }
# Step 4: Run the simulation and check for failure
./Vfifo_tb || { echo "Simulation failed"; exit 1; }
# Step 5: Open the waveform in GTKWave
gtkwave dump.vcd

b) Make the script executable:
chmod +x run.sh
