Building and Verifying a Simple Asynchronous FIFO

1.  Asynchronous fifo
A 4×8-bit asynchronous FIFO stores data on wr_clk and outputs data on rd_clk. Full and empty flags are generated by comparing synchronized read and write pointers.

Create a file : async_fifo.v

module async_fifo #(
    parameter DEPTH = 4,              // FIFO depth
    parameter ADDR_WIDTH = 2          // Address width for DEPTH=4
) (
    // Write domain
    input  wire             wr_clk,
    input  wire             wr_en,
    input  wire [7:0]       wr_data,
    output wire             full,

    // Read domain
    input  wire             rd_clk,
    input  wire             rd_en,
    output reg  [7:0]       rd_data,
    output wire             empty,
    // Global reset (active-low)
    input  wire             rst_n
);
    reg [7:0] mem[0:DEPTH-1];
    // Pointers with extra MSB for wrap detection
    reg [ADDR_WIDTH:0] wr_ptr = 0;
    reg [ADDR_WIDTH:0] rd_ptr = 0;
    // Write logic
    always @(posedge wr_clk or negedge rst_n) begin
        if (!rst_n)
            wr_ptr <= 0;
        else if (wr_en && !full) begin
            mem[wr_ptr[ADDR_WIDTH-1:0]] <= wr_data;
            wr_ptr <= wr_ptr + 1;
        end
    end
    // Read logic
    always @(posedge rd_clk or negedge rst_n) begin
        if (!rst_n) begin
            rd_ptr  <= 0;
            rd_data <= 8'h00;
        end else if (rd_en && !empty) begin
            rd_data <= mem[rd_ptr[ADDR_WIDTH-1:0]];
            rd_ptr  <= rd_ptr + 1;
        end
    end
    // Full flag
  assign full  = (wr_ptr[ADDR_WIDTH] !=rd_ptr[ADDR_WIDTH]) &&(wr_ptr[ADDR_WIDTH-1:0] == rd_ptr[ADDR_WIDTH-1:0]);
    // Empty flag
    assign empty = (wr_ptr == rd_ptr);
endmodule

2. Testbench 
A testbench generates independent write and read clocks and applies simple write/read sequences to the FIFO. It verifies operation at different clock speeds and records waveforms for later analysis.

Create a file :async_fifo_tb.v

`timescale 1ns/1ps
module async_fifo_tb;
    reg wr_clk = 0;
    reg rd_clk = 0;
    reg rst_n  = 0;
    reg        wr_en = 0;
    reg [7:0]  wr_data = 0;
    wire       full;
    reg        rd_en = 0;
    wire [7:0] rd_data;
    wire       empty;
    // DUT Instance
    async_fifo uut (
        .wr_clk(wr_clk), .wr_en(wr_en), .wr_data(wr_data), .full(full),
        .rd_clk(rd_clk), .rd_en(rd_en), .rd_data(rd_data), .empty(empty),
        .rst_n(rst_n)
    );
    // Clock generators
    always #4 wr_clk = ~wr_clk; // 125 MHz write clock
    always #7 rd_clk = ~rd_clk; // ~71 MHz read clock

    initial begin
        $dumpfile("async_fifo.vcd");
        $dumpvars(0, async_fifo_tb);
        // Hold reset for a short time, release
        #5  rst_n = 1;
        // Write 4 values
        repeat (4) begin
            @(posedge wr_clk);
            if (!full) begin
                wr_en   <= 1;
                wr_data <= wr_data + 1;
            end
        end
        wr_en <= 0;
        // Read 4 values
        repeat (4) begin
            @(posedge rd_clk);
            if (!empty) rd_en <= 1;
            @(posedge rd_clk);
            rd_en <= 0;
        end
        #50 $finish;
    end
endmodule
