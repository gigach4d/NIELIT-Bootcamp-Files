Clock Gating Techniques

Lab 2.1: Positive latch gating

a) Create file for design :pos_edge_gating.v
// Positive latch based clock gating (unsafe for real designs)

module pos_latch_gating (
    input  wire clk,
    input  wire en,
    input  wire rst,
    output reg  [3:0] count,
    output wire gated_clk
);
    reg en_latch;

    // Positive latch: transparent when clk is high
    always @(clk or en) begin
        if (clk)
            en_latch = en;
    end

    assign gated_clk = clk & en_latch;

    always @(posedge gated_clk or posedge rst) begin
        if (rst)
            count <= 4'b0000;
        else
            count <= count + 1;
    end
endmodule

        .count(count), .gated_clk(gated_clk)

        #5  en = 1; #3 en = 0;  // short pulse near rising edge

    end
endmodule

Lab 3: ICG cell model

a) Create a design file: icg_cell_model.v

// Simple behavioral model of an Integrated Clock Gating cell
module icg_cell_model (
    input  wire clk,
    input  wire en,
    input  wire rst,
    output reg  [3:0] count,
    output wire gated_clk,
    output reg  en_latch_out
);
    reg en_latch;
    // Capture enable only when clk is low
    always @(clk or en) begin
        if (!clk)
            en_latch = en;
    end
    assign gated_clk = clk & en_latch;
    always @* en_latch_out = en_latch;

    always @(posedge gated_clk or posedge rst) begin
        if (rst)
            count <= 4'b0000;
        else
            count <= count + 1;
    end
endmodule
